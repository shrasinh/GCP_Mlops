# MLflow Server Setup Commands

# This command creates a firewall rule to allow incoming traffic on port 5000, which is the port used by MLflow tracking server
gcloud compute firewall-rules create allow-mlops \
  --allow tcp:5000 \
  --source-ranges 0.0.0.0/0 \
  --description "Allow access for mlops assignment"

# This command starts the MLflow tracking server on the Vertex AI workbench instance
mlflow server --host 0.0.0.0 --port 5000

# This command retrieves the external (public) IP address of Vertex AI workbench instance
# The output will be just the IP address (e.g., 34.123.45.67)
# Access MLflow UI at: http://[IP_ADDRESS]:5000
gcloud compute instances describe mlops-assignment \
  --zone=us-central1-a \
  --format='get(networkInterfaces[0].accessConfigs[0].natIP)'
